# GlobalGameMonitor Sprint 1 — MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working local dashboard that can track mobile games across App Store and Google Play, scrape top charts and app details, monitor Meta Ad Library creatives, and display all data in a table-based UI.

**Architecture:** Next.js App Router monolith with SQLite (WAL mode) via Drizzle ORM. Scrapers are modular classes extending a base scraper with retry/rate-limit logic. API routes trigger scrapers and serve data. Frontend uses shadcn/ui + TanStack Table for data display.

**Tech Stack:** Next.js 14+, TypeScript, SQLite (better-sqlite3), Drizzle ORM, shadcn/ui, Tailwind CSS, TanStack Table, Recharts, Cheerio, google-play-scraper

> **Note:** Playwright and app-store-scraper are deferred to Sprint 2. Meta Ad Library scraper is a stub in Sprint 1.

**Spec:** `docs/superpowers/specs/2026-03-23-global-game-monitor-design.md`

```typescript
export function formatNumber(n: number): string {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1).replace(/\.0$/, "")}M`;
  if (n >= 1_000) return `${(n / 1_000).toFixed(1).replace(/\.0$/, "")}K`;
  return n.toString();
}

export function formatDate(isoDate: string): string {
  return new Date(isoDate).toLocaleDateString("en-US", {
    month: "short",
    day: "numeric",
    year: "numeric",
    timeZone: "UTC",
  });
}

export function parseDownloadRange(range: string): number {
  if (!range) return 0;
  const cleaned = range.replace(/[,+\s]/g, "");
  if (cleaned.endsWith("M")) return parseFloat(cleaned) * 1_000_000;
  if (cleaned.endsWith("K")) return parseFloat(cleaned) * 1_000;
  const num = parseInt(cleaned, 10);
  return isNaN(num) ? 0 : num;
}

export function timeAgo(isoDate: string): string {
  const diff = Date.now() - new Date(isoDate).getTime();
  const minutes = Math.floor(diff / 60_000);
  if (minutes < 60) return `${minutes}m ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.floor(hours / 24);
  return `${days}d ago`;
}
```

```typescript
export interface ScraperConfig {
  name: string;
  category: "market" | "competitor" | "ads" | "community";
  rateLimit: { requests: number; perSeconds: number };
  retryCount: number;
  timeout: number;
}

export interface ScraperResult<T> {
  source: string;
  fetchedAt: Date;
  records: T[];
  errors: string[];
}

export interface ScraperRunResult {
  status: "success" | "error";
  recordsFetched: number;
  error?: string;
  errors: string[];
  durationMs: number;
}

export function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

export abstract class BaseScraper<T> {
  abstract config: ScraperConfig;
  abstract fetch(): Promise<ScraperResult<T>>;
  abstract store(records: T[]): Promise<void>;

  /** Enforce rate limit delay between scraper runs */
  protected async rateLimit(): Promise<void> {
    const { requests, perSeconds } = this.config.rateLimit;
    const delayMs = (perSeconds / requests) * 1000;
    await sleep(delayMs);
  }

  async run(): Promise<ScraperRunResult> {
    const startTime = Date.now();
    const allErrors: string[] = [];

    for (let attempt = 1; attempt <= this.config.retryCount; attempt++) {
      try {
        const result = await this.fetch();
        allErrors.push(...result.errors);

        if (result.records.length > 0) {
          await this.store(result.records);
        }

        return {
          status: "success",
          recordsFetched: result.records.length,
          errors: allErrors,
          durationMs: Date.now() - startTime,
        };
      } catch (err) {
        const errorMsg = err instanceof Error ? err.message : String(err);
        allErrors.push(`Attempt ${attempt}: ${errorMsg}`);

        if (attempt < this.config.retryCount) {
          const backoff = Math.pow(2, attempt - 1) * 1000;
          await sleep(backoff);
        } else {
          return {
            status: "error",
            recordsFetched: 0,
            error: errorMsg,
            errors: allErrors,
            durationMs: Date.now() - startTime,
          };
        }
      }
    }

    // Should not reach here, but TypeScript needs it
    return {
      status: "error",
      recordsFetched: 0,
      error: "Unknown error",
      errors: allErrors,
      durationMs: Date.now() - startTime,
    };
  }
}
```

```typescript
  country: string,
  category: string,

    store: "appstore" as const,
    country,
    category,
    chartType,
    rank: index + 1,
    storeId: item.id,
    name: item.name,
    developer: item.artistName,

export class AppleTopChartsScraper extends BaseScraper<AppleChartEntry> {
  config: ScraperConfig = {
    name: "apple-top-charts",
    category: "market",
    rateLimit: { requests: 5, perSeconds: 10 },
    retryCount: 3,
    timeout: 15000,
  };

  private countries = ["us"];
  private chartTypes = ["top-free", "top-paid"];
  private limit = 100;

  async fetch(): Promise<ScraperResult<AppleChartEntry>> {
    const allEntries: AppleChartEntry[] = [];
    const errors: string[] = [];

    for (const country of this.countries) {
      for (const chartType of this.chartTypes) {
        try {
          const url = `https://rss.applemarketingtools.com/api/v2/${country}/apps/${chartType}/${this.limit}/apps.json`;
          const response = await fetch(url);
          if (!response.ok) {
            errors.push(`Failed to fetch ${country}/${chartType}: ${response.status}`);
            continue;
          }
          const data = await response.json();
          const simpleType = chartType.replace("top-", "");
          const entries = parseAppleRssResponse(data, country.toUpperCase(), "games", simpleType);
          allEntries.push(...entries);
        } catch (err) {
          errors.push(`Error fetching ${country}/${chartType}: ${err instanceof Error ? err.message : String(err)}`);
        }
      }
    }

    return {
      source: "apple-rss",
      fetchedAt: new Date(),
      records: allEntries,
      errors,
    };
  }

  async store(records: AppleChartEntry[]): Promise<void> {
    const today = new Date().toISOString().split("T")[0];

    for (const entry of records) {
      // Upsert app
      const existing = db
        .select()
        .from(apps)
        .where(and(eq(apps.store, "appstore"), eq(apps.storeId, entry.storeId)))
        .get();

      let appId: number;
      if (existing) {
        appId = existing.id;
        db.update(apps)
          .set({ name: entry.name, developer: entry.developer, iconUrl: entry.iconUrl, updatedAt: new Date().toISOString() })
          .where(eq(apps.id, appId))
          .run();
      } else {
        const result = db
          .insert(apps)
          .values({
            store: "appstore",
            storeId: entry.storeId,
            name: entry.name,
            developer: entry.developer,
            category: entry.genre,
            iconUrl: entry.iconUrl,
          })
          .run();
        appId = Number(result.lastInsertRowid);
      }

      // Insert chart entry (ignore duplicates via unique constraint)
      try {
        db.insert(topCharts)
          .values({
            store: entry.store,
            country: entry.country,
            category: entry.category,
            chartType: entry.chartType,
            date: today,
            rank: entry.rank,
            appId,
          })
          .run();
      } catch {
        // Duplicate entry for same day — skip
      }
    }
  }
}
```

```typescript
      db.update(scraperRuns)
        .set({
          finishedAt: new Date().toISOString(),
          status: result.status,
          recordsFetched: result.recordsFetched,
          errorMessage: result.error ?? null,
        })
        .where(eq(scraperRuns.id, runId))
        .run();
    } catch (err) {
      db.update(scraperRuns)
        .set({
          finishedAt: new Date().toISOString(),
          status: "error",
          errorMessage: err instanceof Error ? err.message : String(err),
        })
        .where(eq(scraperRuns.id, runId))
        .run();
    }
  }, 0);
```

```tsx
export function RankingChart({ data, dataKey, label }: { data: Snapshot[]; dataKey: string; label: string }) {
  const chartData = data
    .filter((d: any) => d[dataKey] != null)
    .map((d) => ({ date: d.date, value: (d as any)[dataKey] }))
    .reverse();

  if (chartData.length === 0) {
    return <p className="text-muted-foreground text-sm">No data for {label}</p>;
  }

  return (
    <ResponsiveContainer width="100%" height={200}>
      <LineChart data={chartData}>
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis dataKey="date" tick={{ fontSize: 12 }} />
        <YAxis tick={{ fontSize: 12 }} />
        <Tooltip />
        <Line type="monotone" dataKey="value" stroke="#2563eb" name={label} strokeWidth={2} dot={false} />
      </LineChart>
    </ResponsiveContainer>
  );
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { useParams } from "next/navigation";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { RankingChart } from "@/components/charts/ranking-chart";

  const latestSnapshot = snapshots[0];

          <h2 className="text-2xl font-bold">{app.name}</h2>
          <p className="text-muted-foreground">{app.developer}</p>
          <div className="flex gap-2 mt-1">
            <Badge variant="outline">{app.store}</Badge>
            {app.category && <Badge variant="outline">{app.category}</Badge>}
            {app.isOwnGame && <Badge variant="secondary">Own Game</Badge>}
          </div>
        </div>
      </div>

      {latestSnapshot && (
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
          <Card>
            <CardHeader className="pb-2">
              <CardTitle className="text-sm text-muted-foreground">Rating</CardTitle>

          <Card>
            <CardHeader className="pb-2">
              <CardTitle className="text-sm text-muted-foreground">Rating Count</CardTitle>
            </CardHeader>
            <CardContent>
              <p className="text-2xl font-bold">{latestSnapshot.ratingCount?.toLocaleString() ?? "-"}</p>
            </CardContent>
          </Card>
          <Card>
            <CardHeader className="pb-2">
              <CardTitle className="text-sm text-muted-foreground">Downloads</CardTitle>
            </CardHeader>

        </CardHeader>
        <CardContent>
          <RankingChart data={snapshots} dataKey="downloadsEstimate" label="Downloads" />
        </CardContent>
      </Card>
    </div>
  );
```
